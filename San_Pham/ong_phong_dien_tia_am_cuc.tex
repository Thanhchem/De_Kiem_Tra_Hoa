\begin{tikzpicture}[x=1cm, y=1cm, line cap=round, line join=round]

  % Styles
  \tikzset{
    wire/.style={line width=1.1pt, draw=black!90},
    glass_edge/.style={line width=0.85pt, draw=glassborder},
    pointer/.style={line width=0.55pt, draw=black!85},
    label_txt/.style={font=\fontsize{11.5pt}{14pt}\selectfont, inner sep=1.5pt}

  % -------------------------------------------------------------
  % 1. NGUỒN ĐIỆN (Battery Box at bottom left)
  % -------------------------------------------------------------
  \coordinate (B_bl) at (1.5, -3.2);
  \coordinate (B_br) at (3.5, -3.2);
  \coordinate (B_tr) at (3.5, -2.2);
  \coordinate (B_tl) at (1.5, -2.2);
  
  \coordinate (B_tbl) at ($(B_tl) + (0.42, 0.32)$);
  \coordinate (B_tbr) at ($(B_tr) + (0.42, 0.32)$);
  \coordinate (B_rbr) at ($(B_br) + (0.42, 0.32)$);

  % Top face
  \filldraw[draw=black!80, line width=0.65pt, fill=battop]
    (B_tl) -- (B_tr) -- (B_tbr) -- (B_tbl) -- cycle;

  % Right face
  \filldraw[draw=black!80, line width=0.65pt, fill=batside]
    (B_tr) -- (B_tbr) -- (B_rbr) -- (B_br) -- cycle;

  % Front face
  \filldraw[draw=black!80, line width=0.65pt, fill=batfront]
    (B_bl) -- (B_br) -- (B_tr) -- (B_tl) -- cycle;

  % Battery Terminals (2 metal posts)
  % Left post (Negative) at x = 2.05
  \filldraw[draw=black!80, line width=0.45pt, fill=black!45] (2.05, -2.04) ellipse (0.09cm and 0.04cm);
  \filldraw[draw=black!80, line width=0.45pt, fill=black!25] (1.96, -2.04) rectangle (2.14, -1.86);
  \filldraw[draw=black!80, line width=0.45pt, fill=black!15] (2.05, -1.86) ellipse (0.09cm and 0.04cm);

  % Right post (Positive) at x = 2.95
  \filldraw[draw=black!80, line width=0.45pt, fill=black!45] (2.95, -2.04) ellipse (0.09cm and 0.04cm);
  \filldraw[draw=black!80, line width=0.45pt, fill=black!25] (2.86, -2.04) rectangle (3.04, -1.86);
  \filldraw[draw=black!80, line width=0.45pt, fill=black!15] (2.95, -1.86) ellipse (0.09cm and 0.04cm);

  % 2. ỐNG THỦY TINH (Glass Tube Outline & Body Shading)
  % -------------------------------------------------------------
  % Glass body background shading (Light luminous icy glass)
  \shade[top color=glassblue!70!white, bottom color=glassblue!90!black!15, middle color=white, opacity=0.85]
    (1.8, 1.35) -- (4.6, 1.35)
    to[out=0, in=185] (6.5, 1.48) -- (8.2, 1.50)
    to[out=5, in=195] (11.4, 2.40)

    to[out=165, in=-5] (8.2, 0.10) -- (6.5, 0.12)
    to[out=175, in=0] (4.6, 0.25) -- (1.8, 0.25)
    arc (270:90:0.30cm and 0.55cm) -- cycle;

  % Green glow discharge between Cathode and Anode
  \shade[top color=glowgreen!85!white, bottom color=glowgreen!75!teal!30, middle color=glowgreen!95!white, opacity=0.8]

  % -------------------------------------------------------------

  \filldraw[fill=white, draw=black!85, line width=0.75pt] (0.8, -0.32) circle (0.24cm);
  \node[font=\fontsize{11pt}{12pt}\selectfont\bfseries] at (0.8, -0.32) {$-$};

  \filldraw[fill=white, draw=black!85, line width=0.75pt] (4.6, -0.32) circle (0.24cm);
  \node[font=\fontsize{10.5pt}{12pt}\selectfont\bfseries] at (4.6, -0.32) {$+$};

  % 4. ĐIỆN CỰC VÀ TIA ÂM CỰC
  % -------------------------------------------------------------
  % Cathode Disc (Cực âm) at x = 2.1
  \filldraw[draw=black!90, line width=0.7pt, fill=black!75]
    (2.05, 0.8) ellipse (0.07cm and 0.48cm);
  \filldraw[draw=black!90, line width=0.7pt, fill=black!55]
    (2.17, 0.8) ellipse (0.07cm and 0.48cm);
  \draw[line width=0.7pt, draw=black!90] (2.05, 1.28) -- (2.17, 1.28);
  \draw[line width=0.7pt, draw=black!90] (2.05, 0.32) -- (2.17, 0.32);

  \filldraw[draw=copperorange!90!black, line width=0.8pt, fill=copperorange]
    (4.6, 0.8) ellipse (0.15cm and 0.52cm);

  \filldraw[draw=black!85, fill=white, line width=0.6pt] 
    (4.6, 0.8) ellipse (0.045cm and 0.11cm);

  % Cathode Ray (Tia âm cực) - Emerges from anode aperture to phosphor screen!
  % Faint fluorescent glow halo
  \draw[line width=2.6pt, raygreen!40!white, opacity=0.55] 
    (4.6, 0.8) -- (11.4, 0.8);
  % Sharp laser beam
  \draw[line width=1.35pt, raygreen] 
    (4.6, 0.8) -- (11.4, 0.8);

  % Glass tube upper profile
  \draw[glass_edge] 
    (1.8, 1.35) -- (4.6, 1.35)
    to[out=0, in=185] (6.5, 1.48) -- (8.2, 1.50)
    to[out=5, in=195] (11.4, 2.40);

  % Glass tube lower profile
  \draw[glass_edge] 
    (1.8, 0.25) -- (4.6, 0.25)
    to[out=0, in=175] (6.5, 0.12) -- (8.2, 0.10)
    to[out=-5, in=165] (11.4, -0.80);

  % Left dome outer boundary (convex curved to the left)
  \draw[glass_edge] (1.8, 1.35) arc (90:270:0.30cm and 0.55cm);

  % Phosphor Screen (Màn phosphor) at right end
  \shade[top color=glassblue!60!green!25, bottom color=glassblue!80!green!35, middle color=white!75!green!20, opacity=0.85]
    (11.4, 0.8) ellipse (0.68cm and 1.60cm);
  \draw[glass_edge, line width=0.95pt] 
    (11.4, 0.8) ellipse (0.68cm and 1.60cm);

  \fill[raygreen!45!white, opacity=0.7] (11.4, 0.8) circle (0.13cm);
  \fill[raygreen] (11.4, 0.8) circle (0.06cm);
  \fill[white] (11.4, 0.8) circle (0.02cm);

  \draw[white, line width=1.6pt, opacity=0.75] 
    (1.9, 1.27) -- (4.5, 1.27);
  \draw[white, line width=1.6pt, opacity=0.75] 
    (4.8, 1.27) to[out=0, in=185] (6.5, 1.39) -- (8.2, 1.41) to[out=5, in=195] (11.1, 2.27);
  \draw[white, line width=1.2pt, opacity=0.5] 
    (4.8, 0.33) to[out=0, in=175] (6.5, 0.21) -- (8.2, 0.19) to[out=-5, in=165] (11.1, -0.67);

  \draw[pointer] (lbl_man.north) -- (11.1, 0.05);
